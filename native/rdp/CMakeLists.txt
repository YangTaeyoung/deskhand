if(DH_RDP_FFMPEG)
  # Upstream FindFFmpeg discovers optional installed components too. Supply the
  # exact three-library closure to prevent a cached Homebrew GPL codec from leaking in.
  file(REAL_PATH "${DH_RDP_CODEC_ROOT}" codec_root)
  set(codec_libraries "")
  foreach(component AVCODEC AVUTIL SWSCALE)
    string(TOLOWER "${component}" library)
    set(pc "${codec_root}/lib/pkgconfig/lib${library}.pc")
    if(NOT EXISTS "${pc}" OR NOT EXISTS "${codec_root}/lib/lib${library}.dylib")
      message(FATAL_ERROR "Build the fixed LGPL codec prefix first: ${pc}")
    endif()
    file(REAL_PATH "${codec_root}/lib/lib${library}.dylib" real_library)
    string(FIND "${real_library}" "${codec_root}/" root_position)
    if(NOT root_position EQUAL 0)
      message(FATAL_ERROR "Codec library escapes the verified prefix: ${real_library}")
    endif()
    file(STRINGS "${pc}" version_line REGEX "^Version: ")
    string(REGEX REPLACE "^Version: " "" version "${version_line}")
    set(${component}_INCLUDE_DIRS "${codec_root}/include" CACHE PATH "" FORCE)
    set(${component}_LIBRARIES "${real_library}" CACHE FILEPATH "" FORCE)
    set(${component}_VERSION "${version}" CACHE STRING "" FORCE)
    set(${component}_FOUND TRUE)
    list(APPEND codec_libraries "${real_library}")
  endforeach()
  foreach(component AVDEVICE AVFORMAT AVFILTER POSTPROCESS SWRESAMPLE)
    set(${component}_INCLUDE_DIRS "" CACHE PATH "" FORCE)
    set(${component}_LIBRARIES "" CACHE FILEPATH "" FORCE)
    set(${component}_FOUND FALSE)
  endforeach()
  set(FFMPEG_LIBRARIES "${codec_libraries}" CACHE STRING "" FORCE)
  set(FFMPEG_INCLUDE_DIRS "${codec_root}/include" CACHE STRING "" FORCE)
  set(FFMPEG_DEFINITIONS "" CACHE STRING "" FORCE)
endif()
